hint.slice(-1).toLowerCase() : ""; const sync = () => { clear.hidden = input.value === "" || input.disabled; }; const empty = () => { input.value = ""; input.dispatchEvent(new Event("input", { bubbles: true })); input.dispatchEvent(new Event("change", { bubbles: true })); sync(); input.focus(); }; const onKey = (e) => { if (e.key === "Escape" && input.value) { e.preventDefault(); empty(); } }; const onDoc = (e) => { if (!hint) return; const t = e.target, typing = t && (t.isContentEditable || /^(INPUT|TEXTAREA|SELECT)$/.test(t.tagName)); const hit = hint === "/" ?
