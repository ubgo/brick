// brick.js runs the same command list a brick:js attribute holds, from // code, with the element as "this". The list stays in the markup // (data-on-close), so the script never builds a command from data. The // "navigated" event fires after a client navigation (Options.Navigation), // and brick.on returns the function that stops listening, which a mount // callback may return as its cleanup. brick.mount("Menu", (menu) => brick.on("navigated", () => brick.js(menu, menu.getAttribute("data-on-close"))), );
