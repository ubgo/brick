--- use: Icon: ~/components/Icon.html props: label: {type: string, default: Toggle dark mode} --- {{/* ThemeToggle: switches between light and dark and remembers the choice in localStorage ("bk-theme"). Put this line in to apply a saved choice before first paint: Without JavaScript the button is hidden and the page follows the system. */}} // THEME_KEY is the localStorage key the choice is saved under; the layout's // pre-paint script (see the doc comment) reads the same key, so // change both together. THEMES are the two values of . const THEME_KEY = "bk-theme"; const THEMES = { dark: "dark", light: "light" }; brick.mount("ThemeToggle", (el) => { const root = document.documentElement; // Without an explicit choice the page follows the system. const dark = () => root.dataset.theme ? root.dataset.theme === THEMES.dark : matchMedia("(prefers-color-scheme: dark)").matches; const sync = () => el.setAttribute("aria-pressed", String(dark())); const click = () => { root.dataset.theme = dark() ? THEMES.light : THEMES.dark; try { localStorage.setItem(THEME_KEY, root.dataset.theme); } catch (e) {} // private mode: the choice lasts the page sync(); }; el.hidden = false; sync(); el.addEventListener("click", click); return () => el.removeEventListener("click", click); }) .bk-theme-toggle { display: inline-grid; place-items: center; width: var(--bk-h-md); height: var(--bk-h-md); border: 1px solid var(--bk-border); border-radius: var(--bk-radius); background: var(--bk-surface); color: var(--bk-fg); cursor: pointer; } .bk-theme-toggle:hover { background: var(--bk-accent); } .bk-theme-toggle .bk-theme-toggle__moon, .bk-theme-toggle[aria-pressed=true] .bk-theme-toggle__sun { display: none; } .bk-theme-toggle[aria-pressed=true] .bk-theme-toggle__moon { display: inline; }
