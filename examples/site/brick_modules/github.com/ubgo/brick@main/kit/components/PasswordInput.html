--- use: Icon: ~/components/Icon.html Input: ~/components/Input.html props: size: {type: string, default: md, oneof: [sm, md, lg]} icon: {type: string, default: ""} invalid: {type: bool, default: false} required: {type: bool, default: false} disabled: {type: bool, default: false} showLabel: {type: string, default: Show password} --- {{/* PasswordInput: a password field with a show/hide toggle. The toggle appears with JavaScript and switches the input between type="password" and type="text" (aria-pressed tells a screen reader which). Without JavaScript it is a plain password field. Every attribute but class reaches the . */}} brick.mount("PasswordInput", (el) => { const input = el.querySelector("input"); const btn = el.querySelector(".bk-password__toggle"); const click = () => { const show = input.type === "password"; input.type = show ? "text" : "password"; btn.setAttribute("aria-pressed", String(show)); input.focus(); }; btn.hidden = false; btn.addEventListener("click", click); return () => btn.removeEventListener("click", click); }) .bk-password { width: 100%; } .bk-password__toggle { display: grid; place-items: center; width: 1.6rem; height: 1.6rem; padding: 0; border: 0; border-radius: var(--bk-radius-sm); background: transparent; color: var(--bk-muted-fg); cursor: pointer; } .bk-password__toggle:hover { background: var(--bk-accent); color: var(--bk-fg); } .bk-password__toggle[hidden] { display: none; } .bk-password__toggle .bk-password__off, .bk-password__toggle[aria-pressed=true] .bk-password__on { display: none; } .bk-password__toggle[aria-pressed=true] .bk-password__off { display: inline; }
