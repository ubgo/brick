--- props: accents: {type: list, default: [indigo, blue, emerald, rose, amber, violet, zinc]} size: {type: string, default: md, oneof: [sm, md]} label: {type: string, default: Accent colour} --- {{/* AccentPicker: a row of colour swatches that sets and remembers the choice in localStorage ("bk-accent"). The accents are the ones Theme defines: indigo (the default), blue, emerald, rose, amber, violet and zinc. It is a native radio group, so arrow keys move between swatches. Without JavaScript it is hidden: the page keeps the default accent. Apply a saved choice before first paint with this line in : Every picker follows data-accent, whoever changes it. */}} // DEFAULT_ACCENT is the accent a page without data-accent shows (Theme's // --bk-primary). ACCENT_KEY is the localStorage key the layout's inline // head script reads before first paint; the two must stay in step. const DEFAULT_ACCENT = "indigo"; const ACCENT_KEY = "bk-accent"; brick.mount("AccentPicker", (el) => { const root = document.documentElement; const inputs = [...el.querySelectorAll("input[type=radio]")]; const sync = () => { const cur = root.dataset.accent || DEFAULT_ACCENT; inputs.forEach((i) => { i.checked = i.value === cur; }); }; const change = (e) => { const accent = e.target.value; root.dataset.accent = accent; try { localStorage.setItem(ACCENT_KEY, accent); } catch (err) {} // storage blocked: the choice lasts for this page only }; el.hidden = false; sync(); el.addEventListener("change", change); // Follow data-accent however it changes: another picker, a script, a // server-rendered page swap. const mo = new MutationObserver(sync); mo.observe(root, { attributes: true, attributeFilter: ["data-accent"] }); return () => { el.removeEventListener("change", change); mo.disconnect(); }; }) .bk-accent-picker { display: inline-flex; flex-wrap: wrap; align-items: center; gap: .375rem; } .bk-accent-picker[hidden] { display: none; transition: transform var(--bk-duration), box-shadow var(--bk-duration); } .bk-accent-picker__swatch:hover .bk-accent-picker__dot { transform: scale(1.1); } .bk-accent-picker__input:checked + .bk-accent-picker__dot { box-shadow: 0 0 0 2px var(--bk-bg), 0 0 0 4px var(--bk-fg); } .bk-accent-picker__input:focus-visible + .bk-accent-picker__dot { box-shadow: 0 0 0 2px var(--bk-bg), 0 0 0 4px var(--bk-ring); } .bk-accent-picker__dot--indigo { background: var(--bk-swatch-indigo);
