--- use: settings: ~/data/settings.yaml state: email: ada@example.com language: en timezone: Europe/London birthday: "1815-12-10" saves: 0 saved: false deleting: false --- {{/* SettingsAccount: the Account section — e-mail, language, time zone and birthday, plus a danger zone whose ConfirmDialog runs the deleteAccount action. Deletion is a scheduled state the visitor can undo, so the demo never locks anyone out. */}} LANGS = ["en", "de", "fr", "es", "hi", "ja"] ZONES = ["America/Los_Angeles", "America/Denver", "America/Chicago", "America/New_York", "America/Sao_Paulo", "Europe/London", "Europe/Berlin", "Europe/Athens", "Asia/Dubai", "Asia/Kolkata", "Asia/Singapore", "Asia/Tokyo", "Australia/Sydney"] def save(state, props, fields): email = fields.get("email", "").strip() language = fields.get("language", "") timezone = fields.get("timezone", "") birthday = fields.get("birthday", "").strip() new = dict(state, email = email, language = language, timezone = timezone, birthday = birthday, saved = False) errors = {} at = email.find("@") if at < 1 or "." not in email[at:] or " " in email: errors["email"] = "Enter an e-mail address like ada@example.com." if language not in LANGS: errors["language"] = "Choose a language." if timezone not in ZONES: errors["timezone"] = "Choose a time zone from the list." if birthday and (len(birthday) != 10 or birthday[4] != "-" or birthday[7] != "-"): errors["birthday"] = "Use a date like 1990-04-21." elif birthday and birthday > today(): errors["birthday"] = "Your birthday cannot be in the future." if errors: return [new, errors] return dict(new, saved = True, saves = state["saves"] + 1) def deleteAccount(state, props): return dict(state, deleting = True, saved = False) def undoDelete(state, props): return dict(state, deleting = False, saved = False)
