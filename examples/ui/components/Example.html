export function highlight(pre, lang = "html") {
  const code = pre.querySelector("code") || pre;
  if (code.dataset.hl) return;
  let src = code.textContent;
  const out = document.createDocumentFragment();
  const span = (cls, text) => { const s = document.createElement("span"); s.className = "ex-tk-" + cls; s.textContent = text; out.append(s); };
  if (lang === "code") {
    highlightCode(src, span, out);
    code.textContent = "";
    code.append(out);
    code.dataset.hl = "1";
    return;
  }
  const fm = /^---\n[\s\S]*?\n---\n?/.exec(src);
  if (fm) {
    for (const line of fm[0].split(/(?<=\n)/)) {
      const m = /^(\s*-?\s*)([\w.-]+)(:)(.*\n?)$/.exec(line);
      if (m && line.trim() !== "---") { out.append(m[1]); span("key", m[2]); out.append(m[3]); span("val", m[4]); }
      else span("com", line);
    }
    src = src.slice(fm[0].length);
  }
  let inTag = false;
  while (src) {
    let hit = null;
    for (const [cls, re] of KINDS) {
      if ((cls === "str" || cls === "attr") && !inTag) continue;
      if (cls === "tag" && re.source.startsWith("^\\/?>") && !inTag) continue;
      const m = re.exec(src);
      if (m) { hit = [cls, m[0]]; break; }
    }
    if (hit) {
      span(hit[0], hit[1]);
      if (hit[0] === "tag") inTag = !/>$/.test(hit[1]);
      src = src.slice(hit[1].length);
    } else {
      const next = src.search(inTag ? /[<{"'\/>:@A-Za-z_]/ : /[<{]/);
      const n = next <= 0 ? 1 : next;
      out.append(src.slice(0, n));
      src = src.slice(n);
    }
  }
  code.textContent = "";
  code.append(out);
  code.dataset.hl = "1";
}
brick.mount("Example", (el) => {
  const tabs = [...el.querySelectorAll(":scope > .ex__head [role=tab]")];
  const panels = { preview: el.querySelector(":scope > .ex__preview"), code: el.querySelector(":scope > .ex__code") };
  if (!panels.code) return;
  const select = (name, focus) => {
    tabs.forEach((t) => {
      const on = t.dataset.tab === name;
      t.setAttribute("aria-selected", String(on));
      t.tabIndex = on ? 0 : -1;
      if (on && focus) t.focus();
    });
    panels.preview.hidden = name !== "preview";
    panels.code.hidden = name !== "code";
    if (name === "code") el.querySelectorAll(".ex__code pre").forEach(highlight);
  };
  const onClick = (e) => { const t = e.target.closest("[role=tab]"); if (t && tabs.includes(t)) select(t.dataset.tab, false); };
  const onKey = (e) => {
    const i = tabs.indexOf(document.activeElement);
    if (i < 0 || !["ArrowLeft", "ArrowRight", "Home", "End"].includes(e.key)) return;
    e.preventDefault();
    const j = e.key === "Home" ? 0 : e.key === "End" ? tabs.length - 1 : (i + (e.key === "ArrowRight" ? 1 : -1) + tabs.length) % tabs.length;
    select(tabs[j].dataset.tab, true);
  };
  el.classList.add("ex--js");
  el.addEventListener("click", onClick);
  el.addEventListener("keydown", onKey);
  select("preview", false);
  return () => { el.removeEventListener("click", onClick); el.removeEventListener("keydown", onKey); };
})
